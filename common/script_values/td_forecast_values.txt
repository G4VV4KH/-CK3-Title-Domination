# Shared AI preferences: the same formulas drive real options and the forecast.
# Availability is still enforced by the existing event triggers.
# All values are nonnegative; a truce modifier expresses a peace preference,
# not a claim that defending against an attack breaks a truce.
td_ai_fight_weight_value = {
    value = 50
    if = { limit = { has_trait = brave } add = 50 }
    if = { limit = { has_trait = ambitious } add = 40 }
    if = { limit = { has_trait = craven } subtract = 30 }
    if = { limit = { is_allied_to = scope:td_dominant } subtract = 30 }
    if = { limit = { td_ai_truce_with_dominant_trigger = yes } subtract = 10 }
    if = { limit = { td_ai_allied_to_fighter_trigger = yes } add = 20 }
    min = 0
}
td_ai_ignore_weight_value = {
    value = 30
    if = { limit = { has_trait = craven } add = 50 }
    if = { limit = { any_character_war = { NOT = { using_cb = td_domination_cb } } } add = 30 }
    if = { limit = { td_ai_truce_with_dominant_trigger = yes } add = 10 }
}
td_ai_join_weight_value = {
    value = 10
    if = { limit = { faith = scope:td_dominant.faith } add = 40 }
    if = { limit = { has_trait = content } add = 20 }
    if = { limit = { has_trait = ambitious } subtract = 30 }
    if = { limit = { opinion = { target = scope:td_dominant value >= 50 } } add = 30 }
    if = { limit = { is_allied_to = scope:td_dominant } add = 30 }
    if = { limit = { td_ai_allied_to_fighter_trigger = yes } subtract = 20 }
    min = 0
}
td_ai_religion_weight_value = {
    value = 25
    if = { limit = { has_trait = zealous } add = 30 }
    if = { limit = { has_trait = content } add = 20 }
    if = { limit = { is_allied_to = scope:td_dominant } add = 30 }
    if = { limit = { td_ai_allied_to_fighter_trigger = yes } subtract = 20 }
    min = 0
}

# Per-candidate forecast weights use prospective rank and local predicted sides.
td_forecast_fight_weight_value = {
    value = 0
    if = { limit = { td_forecast_fight_available_trigger = yes } value = td_ai_fight_weight_value }
}
td_forecast_join_weight_value = {
    value = 0
    if = { limit = { td_forecast_join_available_trigger = yes } value = td_ai_join_weight_value }
}
td_forecast_religion_weight_value = {
    value = 0
    if = { limit = { td_forecast_religion_available_trigger = yes } value = td_ai_religion_weight_value }
}
